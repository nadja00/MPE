
#include <LiquidCrystal.h>
#define SELECT 1
#define LEFT 2
#define UP 3
#define DOWN 4
#define RIGHT 5

LiquidCrystal lcd(8, 9, 4, 5, 6, 7);

int brojac = 0;
bool dozvola = true;
void setup() {

 lcd.begin(16, 2);

}
void loop() {

  lcd.home();
  lcd.setCursor(0,0);
  lcd.print(brojac);
  
  byte button = readButton();
  if(button == UP && dozvola){
  	brojac++;
    lcd.clear();
    dozvola = 0;
  }
  else if(button == DOWN && dozvola){
    if(brojac > 0)
  		brojac--;
    lcd.clear();
    dozvola = 0;
  }
  else if(button == SELECT && dozvola){
  	brojac = 0;
    lcd.clear();
    dozvola = 0;
  }
  else if(button != UP && button != DOWN && button != SELECT ){
  	dozvola = 1;
  }
  


}

byte readButton()
{
 int tmp = analogRead(0); 
 if (tmp > 635 && tmp < 645) //SELECT
 	return SELECT;
 if (tmp > 405 && tmp < 415) //LEFT
 	return LEFT;
 if (tmp > 95 && tmp < 105) //UP
 	return UP;
 if (tmp > 252 && tmp < 262) //DOWN
 	return DOWN;
 if (tmp < 5) //RIGHT
 	return RIGHT;
 return 0; //NONE
}
